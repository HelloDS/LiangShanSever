## 1. 索引边界与使用方法

动作发送入口来自当前 Creator TS；共享大厅和模块请求索引来自归档 JS 的直接 `api/c` 字面量。候选字段通过对象键及后续赋值提取，不会可靠推导所有参数类型、必填条件、动态路由或调用方响应字段；嵌套函数可能在多个方法中重复出现。原先关闭/隐藏的接口被索引不代表本次要求开放。

`?` 是 TS 可选标记，不等于服务端在任意阶段都可省略。接口还可能混合本地缓存与网络数据；应沿推送读取处验证。生成 Java DTO 时先检查本附录完整摘录，再追踪调用方并补黄金报文。未明确的枚举、规则与响应字段保持待确认。

```typescript
import type { PdkEnvironment, PdkGameApi, PdkGameId } from './pdk-contract';

export function createPdkGameApi(environment: PdkEnvironment, gameId: PdkGameId): PdkGameApi {
    const methods: PdkGameApi = {
        ready() { environment.send({ tag: 'ready' }); },
        chupai(_player, cardType, ...play) {
            const [cards, rounds] = play;
            environment.send({ tag: 'chupai', outCards: cards, rounds, opCardType: cardType });
        },
        buchu() {
            // Only the archived three-player variant sends from this entry point.
            if (gameId === '6014') environment.send({ tag: 'buchu' });
        },
        qiangZhuang() { environment.send({ tag: 'qiangZhuang' }); },
        buQiangZhuang() { environment.send({ tag: 'buQiangZhuang' }); },
        applyAuto(type) { environment.send({ tag: 'applyAuto', type }); },
    };
    if (gameId !== '6014') methods.teShuPaiXin = type => { environment.send({ tag: 'teShuPaixin', teShuPaiXin: type }); };
    if (gameId === '6008' || gameId === '6073') methods.sendUse3D = () => { environment.send({ tag: 'use3D' }); };
    return methods;
}
```

### table-games/mahjong/mahjong-api.ts

[查看源码](../Client/creator/assets/runtime/scripts/games/table-games/mahjong/mahjong-api.ts)；SHA-256：`60558f9a5b0ba640da9a2335e50e0704712c03f0f21bb09728e7f92eb2e66eb2`。

```typescript
import { tableCall, tableProperty, tableRecord } from '../table-runtime';
import type { TableUid, TableMessage, TableNamespace, TableRuntime } from '../table-runtime';

export function createMahjongApi(api: TableNamespace, namespace: TableNamespace, runtime: TableRuntime): object {
    return {
        ready() { runtime.send({ tag: 'ready' }); },
        chupai(mjVal: number, ting: boolean | null = null) {
            const message: TableMessage = { tag: 'chupai', mjVal };
            if (ting != null) message.ting = ting;
            runtime.send(message);
        },
        lacking(lackType: number) { runtime.send({ tag: 'lacking', lackType }); },
        operating(operatingType: number, chiMjs: number[] | null = null) {
            const message: TableMessage = { tag: 'operating', operatingType };
            if (operatingType == 1 && chiMjs != null) message.chiMjs = chiMjs;
            runtime.send(message);
        },
        zimo() { runtime.send({ tag: 'zimo' }); },
        qiangganghu(type: number | string) { runtime.send({ tag: 'qiangganghu', type }); },
        callGangApi(card: number) {
            const data = tableRecord(namespace.MajiangGameData);
            tableCall(data, 'getCurrentGame');
            const hand = tableCall(data, 'getMjs');
            const counts = tableCall(tableRecord(namespace.xxMajiangConstant), 'array_values_count', [hand]);
            tableCall(api, tableProperty(counts, card) == 4 ? 'anGang' : 'gangSelf', [card]);
        },
        gangSelf(mjVal: number) { runtime.send({ tag: 'gangSelf', mjVal }); },
        anGang(mjVal: number) { runtime.send({ tag: 'anGang', mjVal }); },
        ting(remove: number[], tingVal: number[], tingTarget: TableUid) {
            runtime.send({ tag: 'ting', remove, tingVal, tingTarget });
        },
        tingOperate(type: number | string = 0) {
            const message: TableMessage = { tag: 'tingOperate' };
            if (type != 0) message.type = type;
            runtime.send(message);
        },
        applyAuto(_type: number | string = 1) { runtime.send({ tag: 'applyAuto', type: 0 }); },
        applyAutoHu() { runtime.send({ tag: 'applyAutoHu' }); },
        applyTangpai(type: number | string = 0) { runtime.send({ tag: 'applyTang', type }); },
        tangpai(cards: number[], type: number | string = 0) {
            const message: TableMessage = { tag: 'tangOperate', cards };
            if (type != 0) message.type = type;
            runtime.send(message);
        },
        touThrOperate(cards: number[]) { runtime.send({ tag: 'touThrOperate', cards }); },
        changeThree(cards: number[]) { runtime.send({ tag: 'changeThree', cards }); },
    };
}
```
